extends CanvasLayer
class_name NLBHUD
## Debug HUD only: HP, state line, cooldown bar, hint, message.

var _hp: Label
var _state: Label
var _bar: ProgressBar
var _msg: Label

func _ready() -> void:
	layer = 10
	add_to_group("hud")
	var root := Control.new()
	root.set_anchors_preset(Control.PRESET_FULL_RECT)
	root.mouse_filter = Control.MOUSE_FILTER_IGNORE
	add_child(root)
	_hp = _label(root, Vector2(16, 12), 20)
	_state = _label(root, Vector2(16, 40), 22)
	_bar = ProgressBar.new()
	_bar.position = Vector2(16, 72)
	_bar.size = Vector2(220, 16)
	_bar.min_value = 0.0
	_bar.max_value = 1.0
	_bar.show_percentage = false
	root.add_child(_bar)
	_msg = _label(root, Vector2(16, 96), 18)
	var hint := _label(root, Vector2(16, 660), 15)
	hint.text = "A/D move | Space jump | Shift/L dash (1 in air) | J/Z/LMB attack | K/X/E/RMB BLINK | R restart"

func _label(parent: Control, pos: Vector2, size: int) -> Label:
	var l := Label.new()
	l.position = pos
	l.add_theme_font_size_override("font_size", size)
	parent.add_child(l)
	return l

func set_hp(cur: int, maximum: int) -> void:
	_hp.text = "HP: %d/%d" % [cur, maximum]

func set_state(is_b: bool, transitioning: bool, cd_frac: float) -> void:
	if transitioning:
		_state.text = "BLINK..."
		_state.add_theme_color_override("font_color", Color.WHITE)
	elif is_b:
		_state.text = "STATE B — hazard OFF / platform GONE / ground tighter"
		_state.add_theme_color_override("font_color", Color(1.0, 0.65, 0.35))
	else:
		_state.text = "STATE A — platform UP / hazard ON / ranged high"
		_state.add_theme_color_override("font_color", Color(0.5, 0.9, 1.0))
	_bar.value = 1.0 - cd_frac

func set_message(t: String) -> void:
	_msg.text = t
